function constant(t, intensity)
    intensity
end

model bg(Dr, Do, Cr, Co, E, Estar, Ar, Ao)
    var species Estar, Cr, Do, Ar
    const species E, Co, Dr, Ao
    ke = 1; kc = 1; kd = 1; ka = 1;
    Edrain : Estar -> E  ; ke * Estar
    Cdrain : Cr    -> Co ; kc * Cr
    Ddrain : Do    -> Dr ; kd * Do
    Adrain : Ar    -> Ao ; ka * Ar
end

model ph(gam, P, Pstar, Ps)
    var species P, Pstar, Ps
    kd = 1; kcsf = 1; kcsb = 1; krc = 1;
    photo  : P           -> Pstar           ; gam * P
    decay  : Pstar       -> P               ; kd * Pstar
    csf    : Pstar       -> Ps              ; kcsf * Pstar
    csb    : Ps          -> Pstar           ; kcsb * Ps
    recomb : Ps          -> P               ; krc * Ps
end

model opA(gam1, gam2)
   var species P1, P1st, P1s, P1o, P2, P2s, P2st, P2o, Do, Cr, Est, Ar, Qr
   const species Dr, Co, E, Ao, Qo
   kar = 1; kpo = 1; kco = 1
   PC   : ph(gam1, P1, P1st, P1s)
   Ared : P1s + Ao + E -> P1o + Ar + Est ; kar * P1s * Ao * E
   Dox  : Dr + P1o     -> Do + P1        ; kpo * Dr * P1o
   Cred : Ar + Co      -> Ao + Cr        ; kco * Ar * Co
   B    : bg(Dr, Do, Cr, Co, E, Estar, Ar, Ao)
end

model opB(gam1, gam2)

   Cred  : Ar + Co      -> Ao + Cr        ; kco * Ar * Co

   B     : bg(Dr, Do, Cr, Co, E, Estar, Ar, Ao)

model opC(gam1, gam2)

   Cred  : Ar + Co + Est  -> Ao + Cr + E    ; kco * Ar * Co * Est
   Credb : Ao + Cr + E    -> Ar + Co + Est  ; kco * Ao * Cr * E
   B     : bg(Dr, Do, Cr, Co, E, Estar, Ar, Ao)

model opD(gam1, gam2)
   var species P1, P1st, P1s, P1o, P2, P2s, P2st, P2o, Do, Cr, Est, Ar, Qr
   const species Dr, Co, E, Ao, Qo
   kar = 1; kpo = 1; kco = 1
   PC   : ph(gam1, P1, P1st, P1s)
   Qred : P1s + Qo + E     -> P1o + Qr + Est   ; kar * P1s * Qo * E
   Qox  : Qr + P1o         -> Qo + P1          ; kpo * Qr * P1o
   Cred : Do + Cr + 3 Est  -> Dr + Co + 3 E    ; kco * Do * Cr * (Est^3);
   B    : bg(Dr, Do, Cr, Co, E, Estar, Ar, Ao)
end
