function constant(t, intensity)
    intensity
end

model bg(Dr, Do, Cr, Co, E, Est)
    var species Do, Cr, Est
    const species E, Co, Dr
    ke = 1; kc = 1; kd = 1; ka = 1;
    Edrain : Est -> E  ; ke * Est
    Cdrain : Cr  -> Co ; kc * Cr
    Ddrain : Do  -> Dr ; kd * Do
end

model ph(gam, P, Pst, Ps)
    var species P, Pst, Ps
    kd = 1; kcsf = 1; kcsb = 1; krc = 1;
    photo  : P   -> Pst ; gam  * P
    decay  : Pst -> P   ; kd   * Pst
    csf    : Pst -> Ps  ; kcsf * Pst
    csb    : Ps  -> Pst ; kcsb * Ps
    recomb : Ps  -> P   ; krc  * Ps
end

model opA(gam1, gam2)
   var species P1, P1st, P1s, P1o, P2, P2st, P2s, P2o
   var species Do, Cr, Est, Ao, Ar, Ao, Qr, Qo
   const species Dr, Co, E
   kar = 1; kpo = 1; kco = 1
   PC   : ph(gam1, P1, P1st, P1s)
   Ared : P1s + Ao + E -> P1o + Ar + Est ; kar * P1s * Ao * E
   Dox  : Dr + P1o     -> Do + P1        ; kpo * Dr * P1o
   Cred : Ar + Co      -> Ao + Cr        ; kco * Ar * Co
   B    : bg(Dr, Do, Cr, Co, E, Est)
end
