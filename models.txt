# Here are a bunch of antimony models which correspond to our various
# schemes of oxygenic photosynthesis. The idea is that they each expose
# the various rate constants that we set in main.py and then we only
# have to use one f-string of a few lines there.

function square_wave(t, period, intensity)
    intensity * (1 - (floor(t / period) - 2 * floor(floor(t / period) / 2)))
end

model photoexcitation(gam, P, Ps)
    P -> Ps; gam * P
end

model respiration(kr, kb, Do, Dr, Co, Cr, E, Estar)
    fw: Do + Cr + 5 E -> $Dr + $Co + 5 Estar; kr*Do*Cr*(E^5);
    bw: $Dr + $Co + 5 Estar -> Do + Cr + 5 E; kb*(Estar^5);
end

model storage(kd, kn, Cr, Cstore)
    # if we want a day-night storage release cycle we can do
    # gam := square_wave(t, period, 1)
    # and then replace kd -> kd * gam; kn -> kn * (1 - gam)
    Cr -> Cstore; kd * Cr
    Cstore -> Cr; kn * Cstore
end

model fuckoff(piss, shit, vomit)
    shit -> vomit; piss * shit * Cr
end

model scheme_1(gamma, k1, k2, kr, kb, k_m, kd, kn)
    // Reactions:
    P: photoexcitation(gamma, P, Ps)
    Cred: Ps + $Co -> Po + Cr; k1 * Ps
    Dox: $Dr + Po -> Do + P; k2 * Po
    R: respiration(kr, kb, Do, Dr, Co, Cr, E, Estar, n)
    M: maintenance(k_m, Estar, E)
    S: storage(kd, kn, Cr, Cstore)

    // Species initializations:
    P = 1; Ps = 0; Po = 0;
    Cr = 0; Co = 1; Cstore = 0;
    Do = 0; Dr = 1;
    E = 1; Estar = 0;
end
